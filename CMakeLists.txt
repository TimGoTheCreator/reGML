cmake_minimum_required(VERSION 3.24)
project(reGML VERSION 0.2.0 LANGUAGES CXX)
option(REGML_BUILD_TESTS "Build correctness tests" ON)
option(REGML_SANITIZERS "Enable ASan/UBSan on supported Unix toolchains" OFF)
option(REGML_SDL2 "Build the SDL2 window backend" ON)
add_library(regml_core src/core/error.cpp src/formats/package.cpp src/bytecode/decoder.cpp src/bytecode/linker.cpp src/vm/interpreter.cpp src/render/canvas.cpp src/render/assets.cpp src/runtime/demo.cpp src/runtime/game.cpp src/runtime/audio.cpp)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VORBISFILE REQUIRED IMPORTED_TARGET vorbisfile)
find_package(PNG REQUIRED)
target_link_libraries(regml_core PUBLIC PNG::PNG PkgConfig::VORBISFILE)
target_include_directories(regml_core PUBLIC include)
target_compile_features(regml_core PUBLIC cxx_std_20)
set_target_properties(regml_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(regml_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(regml_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()
if(REGML_SANITIZERS)
  if(MSVC OR WIN32)
    message(FATAL_ERROR "This sanitizer configuration requires GCC/Clang on Linux or macOS")
  endif()
  target_compile_options(regml_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(regml_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(regml src/app/main.cpp src/platform/sdl_host.cpp)
target_link_libraries(regml PRIVATE regml_core)
if(REGML_SDL2)
  find_package(SDL2 CONFIG REQUIRED)
  target_link_libraries(regml PRIVATE SDL2::SDL2)
  target_compile_definitions(regml PRIVATE REGML_WITH_SDL2=1 SDL_MAIN_HANDLED)
else()
  target_compile_definitions(regml PRIVATE REGML_WITH_SDL2=0)
endif()
set_target_properties(regml PROPERTIES CXX_EXTENSIONS OFF)
if(REGML_BUILD_TESTS)
  enable_testing()
  add_executable(regml_tests tests/core_tests.cpp)
  target_link_libraries(regml_tests PRIVATE regml_core)
  add_test(NAME core COMMAND regml_tests)
  add_test(NAME cli_help COMMAND regml --help)
  add_test(NAME cli_demo COMMAND regml vm-demo)
  set_tests_properties(cli_demo PROPERTIES PASS_REGULAR_EXPRESSION "Result: 42")
  add_test(NAME cli_bad_command COMMAND regml unknown-command)
  set_tests_properties(cli_bad_command PROPERTIES WILL_FAIL TRUE)
  add_test(NAME fixture_emit COMMAND regml_tests --emit-fixture ${CMAKE_CURRENT_BINARY_DIR}/synthetic-data.win)
  set_tests_properties(fixture_emit PROPERTIES FIXTURES_SETUP synthetic_package)
  add_test(NAME cli_inspect COMMAND regml inspect ${CMAKE_CURRENT_BINARY_DIR}/synthetic-data.win --code --strings)
  add_test(NAME cli_disasm COMMAND regml disasm ${CMAKE_CURRENT_BINARY_DIR}/synthetic-data.win 0)
  set_tests_properties(cli_inspect cli_disasm PROPERTIES FIXTURES_REQUIRED synthetic_package)
  set_tests_properties(cli_inspect PROPERTIES PASS_REGULAR_EXPRESSION "gml_Script_test")
  set_tests_properties(cli_disasm PROPERTIES PASS_REGULAR_EXPRESSION "ret.i")
  add_test(NAME cli_play_headless COMMAND regml play-demo --headless --frames 40 --autoplay)
  set_tests_properties(cli_play_headless PROPERTIES PASS_REGULAR_EXPRESSION "Player: 480, 380")
  if(REGML_SDL2)
    add_test(NAME cli_video COMMAND regml play-demo --hidden --frames 2)
    set_tests_properties(cli_video PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" PASS_REGULAR_EXPRESSION "Frames: 2")
  endif()
endif()
